# 实战案例：天气查询工具

> 45分钟用AI做一个功能完整的天气应用

---

## 🎯 项目目标

创建一个现代化的天气查询工具，包含：
- 实时天气查询
- 7天天气预报
- 城市搜索和收藏
- 天气详细信息（温度、湿度、风速等）
- 美观的天气图标和动画效果
- 支持地理定位

---

## 🛠️ 技术栈选择

```bash
# 前端：HTML + CSS + JavaScript
# API：OpenWeatherMap（免费天气API）
# 地理：浏览器Geolocation API
# 动画：CSS动画 + 图标库
```

**选择理由：** API简单易用，适合初学者学习API调用

---

```html
    <script>
        // OpenWeatherMap API配置
        const API_KEY = 'YOUR_API_KEY_HERE'; // 替换为你的API key
        const API_BASE = 'https://api.openweathermap.org/data/2.5';
        
        // DOM元素
        const cityInput = document.getElementById('cityInput');
        const loading = document.getElementById('loading');
        const error = document.getElementById('error');
        const weatherMain = document.getElementById('weatherMain');
        const forecastSection = document.getElementById('forecastSection');
        const savedCities = document.getElementById('savedCities');
        
        // 初始化
        document.addEventListener('DOMContentLoaded', function() {
            loadSavedCities();
            
            // 添加键盘事件
            cityInput.addEventListener('keypress', function(e) {
                if (e.key === 'Enter') {
                    searchWeather();
                }
            });
            
            // 自动定位当前城市
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    position => {
                        const lat = position.coords.latitude;
                        const lon = position.coords.longitude;
                        fetchWeatherByCoords(lat, lon);
                    },
                    err => {
                        console.log('定位失败，显示默认城市');
                        searchWeather('北京');
                    }
                );
            } else {
                searchWeather('北京');
            }
        });
        
        // 搜索天气
        function searchWeather(city = null) {
            const cityToSearch = city || cityInput.value.trim();
            if (!cityToSearch) {
                showError('请输入城市名称');
                return;
            }
            
            showLoading();
            hideError();
            
            // 保存搜索历史
            saveCity(cityToSearch);
            
            fetch(`${API_BASE}/weather?q=${cityToSearch}&appid=${API_KEY}&units=metric&lang=zh_cn`)
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`城市不存在或API错误`);
                    }
                    return response.json();
                })
                .then(data => {
                    displayCurrentWeather(data);
                    return fetch(`${API_BASE}/forecast?q=${cityToSearch}&appid=${API_KEY}&units=metric&lang=zh_cn&cnt=7*8`);
                })
                .then(response => response.json())
                .then(forecastData => {
                    displayForecast(forecastData);
                    hideLoading();
                    weatherMain.style.display = 'block';
                    forecastSection.style.display = 'block';
                })
                .catch(err => {
                    showError(`获取天气信息失败: ${err.message}`);
                    hideLoading();
                });
        }
        
        // 通过坐标获取天气
        function fetchWeatherByCoords(lat, lon) {
            showLoading();
            
            fetch(`${API_BASE}/weather?lat=${lat}&lon=${lon}&appid=${API_KEY}&units=metric&lang=zh_cn`)
                .then(response => response.json())
                .then(data => {
                    displayCurrentWeather(data);
                    return fetch(`${API_BASE}/forecast?lat=${lat}&lon=${lon}&appid=${API_KEY}&units=metric&lang=zh_cn&cnt=7*8`);
                })
                .then(response => response.json())
                .then(forecastData => {
                    displayForecast(forecastData);
                    hideLoading();
                    weatherMain.style.display = 'block';
                    forecastSection.style.display = 'block';
                })
                .catch(err => {
                    console.log('坐标获取天气失败，使用默认城市');
                    searchWeather('北京');
                });
        }
        
        // 显示当前天气
        function displayCurrentWeather(data) {
            document.getElementById('locationName').textContent = data.name + ', ' + data.sys.country;
            document.getElementById('temperature').textContent = Math.round(data.main.temp) + '°C';
            document.getElementById('description').textContent = data.weather[0].description;
            document.getElementById('feelsLike').textContent = Math.round(data.main.feels_like) + '°C';
            document.getElementById('humidity').textContent = data.main.humidity + '%';
            document.getElementById('windSpeed').textContent = data.wind.speed + ' m/s';
            document.getElementById('pressure').textContent = data.main.pressure + ' hPa';
            document.getElementById('visibility').textContent = (data.visibility / 1000).toFixed(1) + ' km';
            document.getElementById('uvIndex').textContent = data.uvi || 'N/A';
            
            // 更新天气图标
            updateWeatherIcon(data.weather[0].icon, 'currentIcon');
        }
        
        // 显示天气预报
        function displayForecast(data) {
            const forecastGrid = document.getElementById('forecastGrid');
            const dailyData = {};
            
            // 处理数据，按天分组
            data.list.forEach(item => {
                const date = new Date(item.dt * 1000);
                const dateStr = date.toLocaleDateString('zh-CN');
                
                if (!dailyData[dateStr]) {
                    dailyData[dateStr] = {
                        temps: [],
                        icons: [],
                        descriptions: [],
                        date: date
                    };
                }
                
                dailyData[dateStr].temps.push(item.main.temp);
                dailyData[dateStr].icons.push(item.weather[0].icon);
                dailyData[dateStr].descriptions.push(item.weather[0].description);
            });
            
            // 显示前7天
            let forecastHTML = '';
            let dayCount = 0;
            
            Object.keys(dailyData).sort().forEach(dateStr => {
                if (dayCount >= 7) return;
                
                const dayData = dailyData[dateStr];
                const avgTemp = dayData.temps.reduce((a, b) => a + b, 0) / dayData.temps.length;
                const dayName = dayData.date.toLocaleDateString('zh-CN', { weekday: 'long' });
                
                forecastHTML += `
                    <div class="forecast-day">
                        <div class="day-name">${dayName}</div>
                        <div class="day-icon">
                            <i class="fas ${getWeatherIcon(dayData.icons[0])}"></i>
                        </div>
                        <div class="day-temp">${Math.round(avgTemp)}°C</div>
                        <div class="day-desc">${dayData.descriptions[0]}</div>
                    </div>
                `;
                
                dayCount++;
            });
            
            forecastGrid.innerHTML = forecastHTML;
        }
        
        // 更新天气图标
        function updateWeatherIcon(iconCode, elementId) {
            const iconElement = document.getElementById(elementId);
            const iconMap = {
                '01d': 'fa-sun',
                '01n': 'fa-moon',
                '02d': 'fa-cloud-sun',
                '02n': 'fa-cloud-moon',
                '03d': 'fa-cloud',
                '03n': 'fa-cloud',
                '04d': 'fa-cloud',
                '04n': 'fa-cloud',
                '09d': 'fa-cloud-showers-heavy',
                '09n': 'fa-cloud-showers-heavy',
                '10d': 'fa-cloud-sun-rain',
                '10n': 'fa-cloud-moon-rain',
                '11d': 'fa-bolt',
                '11n': 'fa-bolt',
                '13d': 'fa-snowflake',
                '13n': 'fa-snowflake',
                '50d': 'fa-smog',
                '50n': 'fa-smog'
            };
            
            const iconClass = iconMap[iconCode] || 'fa-question';
            iconElement.className = 'fas ' + iconClass;
        }
        
        // 获取天气图标类名
        function getWeatherIcon(iconCode) {
            const iconMap = {
                '01d': 'fa-sun',
                '01n': 'fa-moon',
                '02d': 'fa-cloud-sun',
                '02n': 'fa-cloud-moon',
                '03d': 'fa-cloud',
                '03n': 'fa-cloud',
                '04d': 'fa-cloud',
                '04n': 'fa-cloud',
                '09d': 'fa-cloud-showers-heavy',
                '09n': 'fa-cloud-showers-heavy',
                '10d': 'fa-cloud-sun-rain',
                '10n': 'fa-cloud-moon-rain',
                '11d': 'fa-bolt',
                '11n': 'fa-bolt',
                '13d': 'fa-snowflake',
                '13n': 'fa-snowflake',
                '50d': 'fa-smog',
                '50n': 'fa-smog'
            };
            
            return iconMap[iconCode] || 'fa-question';
        }
        
        // 快速搜索
        function quickSearch(city) {
            cityInput.value = city;
            searchWeather(city);
        }
        
        // 获取当前位置天气
        function getCurrentLocation() {
            if (navigator.geolocation) {
                loading.style.display = 'block';
                navigator.geolocation.getCurrentPosition(
                    position => {
                        const lat = position.coords.latitude;
                        const lon = position.coords.longitude;
                        fetchWeatherByCoords(lat, lon);
                    },
                    err => {
                        showError('无法获取您的位置信息');
                        hideLoading();
                    }
                );
            } else {
                showError('您的浏览器不支持地理定位');
            }
        }
        
        // 保存城市
        function saveCity(city) {
            let savedCities = JSON.parse(localStorage.getItem('savedCities')) || [];
            if (!savedCities.includes(city)) {
                savedCities.push(city);
                localStorage.setItem('savedCities', JSON.stringify(savedCities));
                loadSavedCities();
            }
        }
        
        // 加载保存的城市
        function loadSavedCities() {
            const savedCitiesList = JSON.parse(localStorage.getItem('savedCities')) || [];
            savedCities.innerHTML = '';
            
            if (savedCitiesList.length === 0) {
                savedCities.innerHTML = '<span style="color: #999;">暂无收藏的城市</span>';
                return;
            }
            
            savedCitiesList.forEach(city => {
                const cityElement = document.createElement('div');
                cityElement.className = 'saved-city';
                cityElement.innerHTML = `
                    ${city}
                    <span class="remove" onclick="removeCity('${city}', event)">×</span>
                `;
                cityElement.onclick = (e) => {
                    if (e.target.classList.contains('remove')) return;
                    quickSearch(city);
                };
                savedCities.appendChild(cityElement);
            });
        }
        
        // 删除城市
        function removeCity(city, event) {
            event.stopPropagation();
            let savedCities = JSON.parse(localStorage.getItem('savedCities')) || [];
            savedCities = savedCities.filter(c => c !== city);
            localStorage.setItem('savedCities', JSON.stringify(savedCities));
            loadSavedCities();
        }
        
        // 显示加载状态
        function showLoading() {
            loading.style.display = 'block';
            weatherMain.style.display = 'none';
            forecastSection.style.display = 'none';
        }
        
        // 隐藏加载状态
        function hideLoading() {
            loading.style.display = 'none';
        }
        
        // 显示错误
        function showError(message) {
            error.textContent = message;
            error.style.display = 'block';
```

```javascript
// API限流
const apiRate = {
    lastCall: 0,
    minInterval: 1000 // 1秒
};

function rateLimitedFetch(url, options = {}) {
    const now = Date.now();
    if (now - apiRate.lastCall < apiRate.minInterval) {
        const delay = apiRate.minInterval - (now - apiRate.lastCall);
        return new Promise(resolve => {
            setTimeout(() => {
                apiRate.lastCall = Date.now();
                fetch(url, options).then(resolve);
            }, delay);
        });
    }
    
    apiRate.lastCall = now;
    return fetch(url, options);
}

// 使用限流API
function searchWeather(city = null) {
    const cityToSearch = city || cityInput.value.trim();
    if (!cityToSearch) {
        showError('请输入城市名称');
        return;
    }
    
    showLoading();
    
    rateLimitedFetch(`${API_BASE}/weather?q=${cityToSearch}&appid=${API_KEY}&units=metric&lang=zh_cn`)
        .then(response => {
            if (!response.ok) {
                throw new Error(`城市不存在或API错误 (${response.status})`);
            }
            return response.json();
        })
        .then(data => {
            displayCurrentWeather(data);
            return rateLimitedFetch(`${API_BASE}/forecast?q=${cityToSearch}&appid=${API_KEY}&units=metric&lang=zh_cn&cnt=7*8`);
        })
        .then(response => response.json())
        .then(forecastData => {
            displayForecast(forecastData);
            hideLoading();
            weatherMain.style.display = 'block';
            forecastSection.style.display = 'block';
        })
        .catch(err => {

}
```

```javascript
// 数据缓存
const weatherCache = {
    data: {},
    maxAge: 5 * 60 * 1000, // 5分钟缓存
    get: function(city) {
        const cached = this.data[city];
        if (cached && Date.now() - cached.timestamp < this.maxAge) {
            return cached.data;
        }
        return null;
    },
    set: function(city, data) {
        this.data[city] = {
            data: data,
            timestamp: Date.now()
        };
    },
    clear: function() {
        this.data = {};
    }
};

// 带缓存的天气查询
function searchWeatherWithCache(city = null) {
    const cityToSearch = city || cityInput.value.trim();
    if (!cityToSearch) {
        showError('请输入城市名称');
        return;
    }
    
    // 检查缓存
    const cachedData = weatherCache.get(cityToSearch);
    if (cachedData) {
        console.log('使用缓存数据');
        displayCurrentWeather(cachedData.current);
        displayForecast(cachedData.forecast);
        hideLoading();
        weatherMain.style.display = 'block';
        forecastSection.style.display = 'block';
        return;
    }
    
    // 获取新数据
    fetchWeatherData(cityToSearch);
}

// 保存数据到缓存
function fetchWeatherData(city) {
    rateLimitedFetch(`${API_BASE}/weather?q=${city}&appid=${API_KEY}&units=metric&lang=zh_cn`)
        .then(response => response.json())
        .then(currentData => {
            weatherCache.set(city, { current: currentData });
            
            return rateLimitedFetch(`${API_BASE}/forecast?q=${city}&appid=${API_KEY}&units=metric&lang=zh_cn&cnt=7*8`);
        })
        .then(response => response.json())
        .then(forecastData => {
            weatherCache.set(city, { current: currentData, forecast: forecastData });
            displayCurrentWeather(currentData);
            displayForecast(forecastData);
            hideLoading();
            weatherMain.style.display = 'block';
            forecastSection.style.display = 'block';
        });
}
```
