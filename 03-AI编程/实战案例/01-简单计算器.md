```javascript
let calculationHistory = [];

function calculate() {
    try {
        let expression = currentInput.replace(/×/g, '*');
        let result = eval(expression);
        
        // 添加到历史记录
        let historyItem = {
            expression: currentInput,
            result: result.toString()
        };
        calculationHistory.unshift(historyItem);
        
        // 只保留最近10条
        if (calculationHistory.length > 10) {
            calculationHistory = calculationHistory.slice(0, 10);
        }
        
        currentInput = result.toString();
        shouldResetDisplay = true;
        updateDisplay();
        updateHistory();
    } catch (error) {
        currentInput = 'Error';
        shouldResetDisplay = true;
        updateDisplay();
    }
}

function updateHistory() {
    let historyList = document.getElementById('historyList');
    historyList.innerHTML = '';
    
    calculationHistory.forEach(item => {
        let historyItem = document.createElement('div');
        historyItem.className = 'history-item';
        historyItem.textContent = `${item.expression} = ${item.result}`;
        historyItem.onclick = () => {
            currentInput = item.result;
            updateDisplay();
        };
        historyList.appendChild(historyItem);
    });
}

function clearHistory() {
    calculationHistory = [];
    updateHistory();
}
```

```javascript
// 键盘事件监听
document.addEventListener('keydown', function(event) {
    const key = event.key;
    
    // 数字键
    if (key >= '0' && key <= '9') {
        appendNumber(key);
    }
    // 运算符
    else if (key === '+') appendOperator('+');
    else if (key === '-') appendOperator('-');
    else if (key === '*') appendOperator('*');
    else if (key === '/') appendOperator('/');
    else if (key === 'Enter' || key === '=') calculate();
    else if (key === 'Escape' || key === 'c' || key === 'C') clearDisplay();
    else if (key === 'Backspace') deleteLast();
    else if (key === '.') appendNumber('.');
});
```

```css
    isDarkMode = !isDarkMode;
    document.body.className = isDarkMode ? 'dark' : 'light';

    localStorage.setItem('theme', isDarkMode ? 'dark' : 'light');
}

    if (savedTheme === 'dark') {
        isDarkMode = true;
```

#### 1. 除零错误处理
```javascript
function calculate() {
    try {
        let expression = currentInput.replace(/×/g, '*');
        
        // 检查除零
        if (expression.includes('/0') || expression.includes('/0.') || expression.includes('/+0')) {
            currentInput = 'Error: Division by zero';
            shouldResetDisplay = true;
            updateDisplay();
            return;
        }
        
        let result = eval(expression);
        currentInput = result.toString();
        shouldResetDisplay = true;
        updateDisplay();
    } catch (error) {
        currentInput = 'Error';
        shouldResetDisplay = true;
        updateDisplay();
    }
}
```

#### 2. 输入验证
```javascript
function appendNumber(num) {
    if (shouldResetDisplay) {
        currentInput = '';
        shouldResetDisplay = false;
    }
    
    // 限制输入长度
    if (currentInput.length >= 20) {
        alert('输入过长');
        return;
    }
    
    if (currentInput === '0' && num !== '.') {
        currentInput = num;
    } else {
        currentInput += num;
    }
    updateDisplay();
}
```

#### 3. 大数处理
```javascript
function calculate() {
    try {
        let expression = currentInput.replace(/×/g, '*');
        let result = eval(expression);
        
        // 处理大数
        if (result > Number.MAX_SAFE_INTEGER) {
            currentInput = result.toExponential(6);
        } else if (result < Number.MIN_SAFE_INTEGER) {
            currentInput = result.toExponential(6);
        } else {
            currentInput = result.toString();
        }
        
        shouldResetDisplay = true;
        updateDisplay();

        currentInput = 'Error';
        shouldResetDisplay = true;
        updateDisplay();
    }
```
