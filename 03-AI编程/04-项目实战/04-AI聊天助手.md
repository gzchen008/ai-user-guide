## 📖 项目介绍

在这个项目中，我们将创建一个可以与 AI 对话的聊天应用。这是一个非常实用的项目，你可以在它的基础上做各种定制开发。

### 第四步：实现核心逻辑

创建 `app.js`：

```javascript
// 获取 DOM 元素
const chatContainer = document.getElementById('chatContainer');
const messageInput = document.getElementById('messageInput');
const sendBtn = document.getElementById('sendBtn');
const clearBtn = document.getElementById('clearBtn');

// 存储对话历史
let messages = [];

// 初始化
function init() {
    // 绑定事件
    sendBtn.addEventListener('click', sendMessage);
    clearBtn.addEventListener('click', clearChat);
    
    // 回车发送（Shift+回车换行）
    messageInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            sendMessage();
        }
    });
    
    // 自动调整输入框高度
    messageInput.addEventListener('input', autoResize);
}

// 自动调整输入框高度
function autoResize() {
    messageInput.style.height = 'auto';
    messageInput.style.height = Math.min(messageInput.scrollHeight, 150) + 'px';
}

// 发送消息
async function sendMessage() {
    const content = messageInput.value.trim();
    if (!content) return;
    
    // 禁用发送按钮
    sendBtn.disabled = true;
    messageInput.value = '';
    messageInput.style.height = 'auto';
    
    // 添加用户消息
    addMessage('user', content);
    
    // 显示加载动画
    showLoading();
    
    try {
        // 调用 AI API
        const response = await callAI(content);
        
        // 移除加载动画
        hideLoading();
        
        // 添加 AI 回复
        addMessage('assistant', response);
    } catch (error) {
        hideLoading();
        addMessage('assistant', '抱歉，出现了错误：' + error.message);
        console.error('Error:', error);
    }
    
    // 恢复发送按钮
    sendBtn.disabled = false;
    messageInput.focus();
}

// 添加消息到界面
function addMessage(role, content) {
    // 保存到历史
    messages.push({ role, content });
    
    // 创建消息元素
    const messageDiv = document.createElement('div');
    messageDiv.className = `message ${role}`;
    messageDiv.innerHTML = `
        <div class="message-content">${escapeHtml(content)}</div>
    `;
    
    // 添加到容器
    chatContainer.appendChild(messageDiv);
    
    // 滚动到底部
    scrollToBottom();
}

// 显示加载动画
function showLoading() {
    const loadingDiv = document.createElement('div');
    loadingDiv.className = 'message assistant';
    loadingDiv.id = 'loading';
    loadingDiv.innerHTML = `
        <div class="loading">
            <span></span>
            <span></span>
            <span></span>
        </div>
    `;
    chatContainer.appendChild(loadingDiv);
    scrollToBottom();
}

// 隐藏加载动画
function hideLoading() {
    const loading = document.getElementById('loading');
    if (loading) loading.remove();
}

// 调用 AI API
async function callAI(userMessage) {
    const config = getCurrentConfig();
    
    // 构建消息历史
    const apiMessages = [
        { role: 'system', content: CONFIG.systemPrompt },
        ...messages.slice(-10) // 只保留最近 10 条，避免太长
    ];
    
    // 根据不同服务调用不同 API
    if (CONFIG.provider === 'deepseek' || CONFIG.provider === 'openai') {
        return await callOpenAICompatible(config, apiMessages);
    } else if (CONFIG.provider === 'claude') {
        return await callClaude(config, apiMessages);
    }
}

// 调用 OpenAI 兼容的 API（DeepSeek、OpenAI）
async function callOpenAICompatible(config, apiMessages) {
    const response = await fetch(`${config.baseUrl}/chat/completions`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${config.apiKey}`
        },
        body: JSON.stringify({
            model: config.model,
            messages: apiMessages,
            temperature: 0.7,
            max_tokens: 2000
        })
    });
    
    if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error?.message || 'API 调用失败');
    }
    
    const data = await response.json();
    return data.choices[0].message.content;
}

// 调用 Claude API
async function callClaude(config, apiMessages) {
    // Claude API 格式稍有不同
    const response = await fetch(`${config.baseUrl}/messages`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'x-api-key': config.apiKey,
            'anthropic-version': '2023-06-01'
        },
        body: JSON.stringify({
            model: config.model,
            max_tokens: 2000,
            system: CONFIG.systemPrompt,
            messages: apiMessages.filter(m => m.role !== 'system')
        })
    });
    
    if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error?.message || 'API 调用失败');
    }
    
    const data = await response.json();
    return data.content[0].text;
}

// 清空对话
function clearChat() {
    messages = [];
    chatContainer.innerHTML = `
        <div class="welcome-message">
            <p>👋 你好！我是 AI 助手，有什么可以帮你的？</p>
        </div>
    `;
}

// 滚动到底部
function scrollToBottom() {
    chatContainer.scrollTop = chatContainer.scrollHeight;
}

// 转义 HTML（防止 XSS）
function escapeHtml(text) {
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
}

// 启动应用
init();
```

### 1. 流式输出（打字机效果）

修改 `app.js`，添加流式输出：

```javascript
// 流式调用 AI API
async function callAIStream(userMessage) {
    const config = getCurrentConfig();
    const apiMessages = [
        { role: 'system', content: CONFIG.systemPrompt },
        ...messages.slice(-10)
    ];
    
    const response = await fetch(`${config.baseUrl}/chat/completions`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${config.apiKey}`
        },
        body: JSON.stringify({
            model: config.model,
            messages: apiMessages,
            stream: true // 开启流式输出
        })
    });
    
    // 创建消息容器
    const messageDiv = document.createElement('div');
    messageDiv.className = 'message assistant';
    messageDiv.innerHTML = '<div class="message-content"></div>';
    chatContainer.appendChild(messageDiv);
    const contentDiv = messageDiv.querySelector('.message-content');
    
    // 读取流式响应
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    let fullContent = '';
    
    while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        
        const chunk = decoder.decode(value);
        const lines = chunk.split('\n');
        
        for (const line of lines) {
            if (line.startsWith('data: ')) {
                const data = line.slice(6);
                if (data === '[DONE]') continue;
                
                try {
                    const json = JSON.parse(data);
                    const content = json.choices[0]?.delta?.content || '';
                    fullContent += content;
                    contentDiv.textContent = fullContent;
                    scrollToBottom();
                } catch (e) {
                    // 忽略解析错误
                }
            }
        }
    }
    
    return fullContent;
}
```

## 🐛 常见问题

### Q1: API 调用报错 "CORS"

**原因：** 浏览器安全限制，不能直接调用外部 API

**解决方案：**
1. 使用本地代理服务器
2. 或者使用支持 CORS 的 API（DeepSeek 支持）

### Q2: 消息显示乱码

**原因：** 文件编码问题

**解决方案：**
- 确保所有文件都是 UTF-8 编码
- HTML 添加 `<meta charset="UTF-8">`

### Q3: AI 回复太慢

**原因：** 网络或 API 响应慢

**解决方案：**
- 使用流式输出，让用户看到逐步响应
- 选择响应更快的 API 服务
