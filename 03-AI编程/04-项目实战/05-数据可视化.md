# 实战项目五：数据可视化工具

> 难度：⭐⭐ 入门级 | 预计用时：2-3 小时

---

## 📖 项目简介

开发一个**数据可视化工具**，可以上传数据文件，自动生成漂亮的图表。

**你能学到：**
- 数据处理基础
- Chart.js 图表库使用
- 文件上传和读取
- 动态生成图表
- 响应式布局

---

## 🎯 功能目标

### 核心功能
- ✅ 上传 CSV 文件
- ✅ 自动解析数据
- ✅ 生成柱状图
- ✅ 生成折线图
- ✅ 生成饼图
- ✅ 切换图表类型

### 进阶功能
- 🔧 支持更多数据格式
- 🔧 自定义图表颜色
- 🔧 导出图片
- 🔧 数据筛选

## ⚙️ 第三步：编写核心逻辑

创建 `app.js`：

```javascript
// 全局变量
let csvData = [];
let columnNames = [];
let myChart = null;

// ========== 文件上传处理 ==========

// 获取DOM元素
const dropZone = document.getElementById('dropZone');
const fileInput = document.getElementById('fileInput');

// 拖拽事件
dropZone.addEventListener('dragover', (e) => {
    e.preventDefault();
    dropZone.classList.add('dragover');
});

dropZone.addEventListener('dragleave', () => {
    dropZone.classList.remove('dragover');
});

dropZone.addEventListener('drop', (e) => {
    e.preventDefault();
    dropZone.classList.remove('dragover');
    
    const file = e.dataTransfer.files[0];
    if (file) {
        handleFile(file);
    }
});

// 点击上传
fileInput.addEventListener('change', (e) => {
    const file = e.target.files[0];
    if (file) {
        handleFile(file);
    }
});

// ========== 文件处理 ==========

function handleFile(file) {
    // 检查文件类型
    if (!file.name.endsWith('.csv')) {
        alert('请上传 CSV 格式的文件！');
        return;
    }
    
    // 检查文件大小（5MB限制）
    if (file.size > 5 * 1024 * 1024) {
        alert('文件大小不能超过 5MB！');
        return;
    }
    
    // 使用PapaParse解析CSV
    Papa.parse(file, {
        header: true,
        skipEmptyLines: true,
        complete: function(results) {
            csvData = results.data;
            columnNames = results.meta.fields;
            
            displayPreview();
            setupControls();
        },
        error: function(error) {
            alert('文件解析失败：' + error.message);
        }
    });
}

// ========== 数据预览 ==========

function displayPreview() {
    const previewSection = document.getElementById('previewSection');
    const tableHead = document.getElementById('tableHead');
    const tableBody = document.getElementById('tableBody');
    
    // 生成表头
    let headHtml = '<tr>';
    columnNames.forEach(name => {
        headHtml += `<th>${name}</th>`;
    });
    headHtml += '</tr>';
    tableHead.innerHTML = headHtml;
    
    // 生成表体（只显示前10行）
    let bodyHtml = '';
    const displayData = csvData.slice(0, 10);
    displayData.forEach(row => {
        bodyHtml += '<tr>';
        columnNames.forEach(col => {
            bodyHtml += `<td>${row[col] || '-'}</td>`;
        });
        bodyHtml += '</tr>';
    });
    
    // 如果数据超过10行，显示提示
    if (csvData.length > 10) {
        bodyHtml += `<tr><td colspan="${columnNames.length}" style="text-align:center; color:#999;">
            ... 还有 ${csvData.length - 10} 行数据
        </td></tr>`;
    }
    
    tableBody.innerHTML = bodyHtml;
    previewSection.style.display = 'block';
}

// ========== 控制面板 ==========

function setupControls() {
    const labelSelect = document.getElementById('labelColumn');
    const dataSelect = document.getElementById('dataColumn');
    
    // 清空现有选项
    labelSelect.innerHTML = '';
    dataSelect.innerHTML = '';
    
    // 添加列名选项
    columnNames.forEach(name => {
        labelSelect.innerHTML += `<option value="${name}">${name}</option>`;
        dataSelect.innerHTML += `<option value="${name}">${name}</option>`;
    });
    
    // 显示控制面板
    document.getElementById('chartControls').style.display = 'block';
}

// ========== 生成图表 ==========

function generateChart() {
    const chartType = document.getElementById('chartType').value;
    const labelCol = document.getElementById('labelColumn').value;
    const dataCol = document.getElementById('dataColumn').value;
    const chartTitle = document.getElementById('chartTitle').value || '数据图表';
    
    // 提取标签和数据
    const labels = csvData.map(row => row[labelCol]);
    const data = csvData.map(row => {
        const value = parseFloat(row[dataCol]);
        return isNaN(value) ? 0 : value;
    });
    
    // 销毁旧图表
    if (myChart) {
        myChart.destroy();
    }
    
    // 生成颜色
    const colors = generateColors(data.length);
    
    // 创建新图表
    const ctx = document.getElementById('myChart').getContext('2d');
    myChart = new Chart(ctx, {
        type: chartType,
        data: {
            labels: labels,
            datasets: [{
                label: dataCol,
                data: data,
                backgroundColor: chartType === 'line' ? 'rgba(102, 126, 234, 0.2)' : colors,
                borderColor: chartType === 'line' ? '#667eea' : colors.map(c => c.replace('0.7', '1')),
                borderWidth: 2,
                fill: chartType === 'line'
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                title: {
                    display: true,
                    text: chartTitle,
                    font: {
                        size: 18
                    }
                },
                legend: {
                    display: chartType === 'pie' || chartType === 'doughnut' || chartType === 'polarArea'
                }
            },
            scales: chartType === 'pie' || chartType === 'doughnut' || chartType === 'polarArea' ? {} : {
                y: {
                    beginAtZero: true
                }
            }
        }
    });
    
    // 显示图表区域
    document.getElementById('chartSection').style.display = 'block';
}

// 生成随机颜色
function generateColors(count) {
    const baseColors = [
        'rgba(102, 126, 234, 0.7)',
        'rgba(72, 187, 120, 0.7)',
        'rgba(237, 100, 166, 0.7)',
        'rgba(246, 173, 85, 0.7)',
        'rgba(128, 90, 213, 0.7)',
        'rgba(56, 178, 172, 0.7)',
        'rgba(237, 137, 54, 0.7)',
        'rgba(66, 153, 225, 0.7)'
    ];
    
    const colors = [];
    for (let i = 0; i < count; i++) {
        colors.push(baseColors[i % baseColors.length]);
    }
    return colors;
}

// ========== 下载图表 ==========

function downloadChart() {
    if (!myChart) {
        alert('请先生成图表！');
        return;
    }
    
    const link = document.createElement('a');
    link.download = 'chart.png';
    link.href = document.getElementById('myChart').toDataURL('image/png');
    link.click();
}

// ========== 示例数据 ==========

function loadSampleData() {
    // 模拟CSV数据
    const sampleCSV = `月份,销售额,成本,利润
1月,12000,8000,4000
2月,15000,9000,6000
3月,18000,10000,8000
4月,22000,12000,10000
5月,28000,15000,13000
6月,35000,18000,17000`;
    
    Papa.parse(sampleCSV, {
        header: true,
        skipEmptyLines: true,
        complete: function(results) {
            csvData = results.data;
            columnNames = results.meta.fields;
            
            displayPreview();
            setupControls();
            
            // 自动设置标题
            document.getElementById('chartTitle').value = '2026年上半年销售数据';
        }
    });
}
```

## 🐛 常见问题

### Q1: 上传文件后没反应？

**检查：**
- 确认文件是 CSV 格式
- 打开浏览器控制台（F12）看是否有错误
- 检查文件是否包含表头（第一行）

### Q2: 图表显示空白？

**检查：**
- 确认数据列包含数字（不是纯文字）
- 检查 CSV 文件编码是否为 UTF-8
- 查看控制台是否有错误信息

### Q3: 下载图片失败？

**解决：**
```javascript
// 确保在生成图表时添加了这个配置
options: {
    responsive: true,
    maintainAspectRatio: false
}
```

### Q4: 中文显示乱码？

**解决：**
- 用记事本打开 CSV，另存为时选择 UTF-8 编码
- 或使用 Excel 导出时选择"CSV UTF-8"
